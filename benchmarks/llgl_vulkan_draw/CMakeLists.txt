cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKLLGLVulkanDraw LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Audited Windows x64 MSVC draw profile required")
endif()
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../.." root)
set(LLGL_PROBE_BUILD "${root}/out/phase3-llgl-vulkan-20261028-2" CACHE PATH "Audited base import libraries")
set(LLGL_PROFILE_BUILD "${root}/out/phase3-llgl-transfer-src-candidate-final" CACHE PATH "Approved transfer-src v1 DLL build")
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
file(GLOB own CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
list(APPEND own "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${root}/benchmarks/llgl_vulkan_upload/Context.hpp" "${root}/benchmarks/vulkan_draw/Frame.hpp"
  "${root}/benchmarks/phase3_comparison/draw.json" "${root}/benchmarks/competitors/sources.lock.json" "${root}/tests/fixtures/SpirvFixtures.hpp"
  "${root}/tests/fixtures/spirv/fullscreen.spvasm" "${root}/tests/fixtures/spirv/red.spvasm" "${root}/tests/fixtures/spirv/green.spvasm")
list(SORT own)
set(identity "profile|llgl-fixed-pin-transfer-src-v1\n")
foreach(p IN LISTS own)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${p}")
  file(SHA256 "${p}" digest)
  file(RELATIVE_PATH name "${root}" "${p}")
  string(APPEND identity "${name}|${digest}\n")
endforeach()
string(SHA256 fingerprint "${identity}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source-identity.txt" "${identity}")
file(SHA256 "${root}/tests/fixtures/SpirvFixtures.hpp" shader_sha)
add_library(llgl_draw_observer SHARED Layer.cpp)
set_target_properties(llgl_draw_observer PROPERTIES OUTPUT_NAME VkLayer_gfx_llgl_draw)
target_include_directories(llgl_draw_observer PRIVATE "${root}/.deps/Vulkan-Headers/include")
target_compile_features(llgl_draw_observer PRIVATE cxx_std_23)
target_compile_options(llgl_draw_observer PRIVATE /W4 /permissive- /EHsc /utf-8)
add_executable(llgl_vulkan_draw main.cpp)
add_dependencies(llgl_vulkan_draw llgl_draw_observer)
target_compile_features(llgl_vulkan_draw PRIVATE cxx_std_23)
target_compile_options(llgl_vulkan_draw PRIVATE /W4 /permissive- /EHsc /utf-8)
target_compile_definitions(llgl_vulkan_draw PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN GFX_LLGL_DRAW_FINGERPRINT="${fingerprint}" GFX_DRAW_SHADER_SHA="${shader_sha}")
target_include_directories(llgl_vulkan_draw PRIVATE "${root}/.deps/LLGL/include" "${root}/.deps/Vulkan-Headers/include")
target_link_libraries(llgl_vulkan_draw PRIVATE "${LLGL_PROBE_BUILD}/build/llgl/build/Release/LLGL.lib" "${LLGL_PROBE_BUILD}/vulkan-1.lib" bcrypt)
add_custom_command(TARGET llgl_vulkan_draw POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${LLGL_PROFILE_BUILD}/build/llgl/build/Release/LLGL.dll" "$<TARGET_FILE_DIR:llgl_vulkan_draw>"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${LLGL_PROFILE_BUILD}/build/llgl/build/Release/LLGL_Vulkan.dll" "$<TARGET_FILE_DIR:llgl_vulkan_draw>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/VkLayer_gfx_llgl_draw.json" CONTENT "{\n\"file_format_version\":\"1.2.0\",\"layer\":{\"name\":\"VK_LAYER_GFX_llgl_draw\",\"type\":\"GLOBAL\",\"library_path\":\"./VkLayer_gfx_llgl_draw.dll\",\"api_version\":\"1.3.290\",\"implementation_version\":\"1\",\"description\":\"Own pass-through LLGL draw observer\",\"functions\":{\"vkNegotiateLoaderLayerInterfaceVersion\":\"gfxDrawNegotiate\"}}}\n")
