cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKVulkanUpload LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This measured profile requires Windows x64 MSVC; Linux remains unverified")
endif()
set(GFX_INSTALLED_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../out/vulkan-presentation-install" CACHE PATH "Frozen installed Vulkan SDK")
set(SDL_BINARY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../out/sdl" CACHE PATH "Reviewed SDL static build")
find_package(GraphicsSDK CONFIG REQUIRED PATHS "${GFX_INSTALLED_SDK}" NO_DEFAULT_PATH)
find_package(SDL3 CONFIG REQUIRED PATHS "${SDL_BINARY_DIR}" NO_DEFAULT_PATH)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
file(READ "${root}/benchmarks/competitors/sources.lock.json" lock)
string(JSON sdl_pin GET "${lock}" sdl commit)
execute_process(COMMAND git -c gc.auto=0 -c maintenance.auto=false -C "${root}/.deps/SDL" rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual STREQUAL sdl_pin)
  message(FATAL_ERROR "SDL checkout differs from reviewed pin")
endif()
file(GLOB own CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
list(APPEND own "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${root}/benchmarks/common/Adapter.hpp" "${root}/benchmarks/common/Environment.cpp" "${root}/benchmarks/competitors/sources.lock.json")
list(SORT own)
set(identity "")
foreach(p IN LISTS own)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${p}")
  file(SHA256 "${p}" sha)
  file(RELATIVE_PATH relative "${root}" "${p}")
  string(APPEND identity "${relative}|${sha}\n")
endforeach()
string(SHA256 fingerprint "${identity}")
add_executable(vulkan_upload main.cpp ../common/Environment.cpp)
target_include_directories(vulkan_upload PRIVATE "${root}/.deps/Vulkan-Headers/include")
target_compile_features(vulkan_upload PRIVATE cxx_std_23)
target_compile_definitions(vulkan_upload PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN GFX_VK_UPLOAD_FINGERPRINT="${fingerprint}" GFX_SDL_PIN="${sdl_pin}")
target_compile_options(vulkan_upload PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(vulkan_upload PRIVATE GraphicsSDK::Core SDL3::SDL3-static dxgi advapi32)
enable_testing()
foreach(sdk graphicssdk sdl)
  add_test(NAME ${sdk}_probe COMMAND vulkan_upload --probe ${sdk} high)
  set_tests_properties(${sdk}_probe PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
endforeach()
