cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKBenchmarks LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The initial measured competitor adapters require Windows D3D12")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLGL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/LLGL" CACHE PATH "Pinned LLGL source")
set(LLGL_BINARY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../out/llgl" CACHE PATH "LLGL build")
find_library(LLGL_LIBRARY NAMES LLGL PATHS "${LLGL_BINARY_DIR}/build/Release" NO_DEFAULT_PATH REQUIRED)
find_package(SDL3 CONFIG REQUIRED PATHS "${CMAKE_CURRENT_SOURCE_DIR}/../out/sdl" NO_DEFAULT_PATH)
option(GFX_BENCH_WITH_SDK "Include the optional GraphicsSDK adapter" ON)
option(GFX_BENCH_WITH_DILIGENT "Include pinned standalone Diligent Core D3D12" OFF)
option(GFX_BENCH_WITH_BGFX "Include pinned single-thread bgfx D3D12" OFF)
option(GFX_BENCH_WITH_FORGE "Include pinned single-thread The Forge D3D12" OFF)
add_executable(gfx_bench main.cpp competitors/LLGL.cpp competitors/SDL.cpp common/Environment.cpp common/Shaders.cpp common/Telemetry.cpp)
if(GFX_BENCH_WITH_FORGE)
  include(competitors/Forge.cmake)
endif()
if(GFX_BENCH_WITH_BGFX)
  include(competitors/Bgfx.cmake)
endif()
if(GFX_BENCH_WITH_DILIGENT)
  set(DILIGENT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/DiligentCore" CACHE PATH "Pinned Diligent Core source")
  set(DILIGENT_NO_DIRECT3D11 ON CACHE BOOL "" FORCE)
  set(DILIGENT_NO_OPENGL ON CACHE BOOL "" FORCE)
  set(DILIGENT_NO_VULKAN ON CACHE BOOL "" FORCE)
  set(DILIGENT_NO_METAL ON CACHE BOOL "" FORCE)
  set(DILIGENT_NO_WEBGPU ON CACHE BOOL "" FORCE)
  set(DILIGENT_NO_ARCHIVER ON CACHE BOOL "" FORCE)
  set(DILIGENT_NO_SUPER_RESOLUTION ON CACHE BOOL "" FORCE)
  set(DILIGENT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(DILIGENT_INSTALL_CORE OFF CACHE BOOL "" FORCE)
  set(DILIGENT_LOAD_PIX_EVENT_RUNTIME OFF CACHE BOOL "" FORCE)
  # Match the existing /O2, /MD, no-LTO benchmark profile. Upstream defaults add /GL and AVX2.
  set(DILIGENT_MSVC_RELEASE_COMPILE_OPTIONS "" CACHE STRING "" FORCE)
  # Override the upstream short NVAPI tag with the full reviewed commit.
  include(FetchContent)
  FetchContent_Declare(nvapi GIT_REPOSITORY https://github.com/NVIDIA/nvapi.git
    GIT_TAG 9296d671e71608d6d6b7749ed93989af4ada8858)
  add_subdirectory("${DILIGENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/diligent" EXCLUDE_FROM_ALL)
  target_sources(gfx_bench PRIVATE competitors/Diligent.cpp)
  target_include_directories(gfx_bench PRIVATE "${DILIGENT_SOURCE_DIR}")
  target_compile_definitions(gfx_bench PRIVATE GFX_BENCH_HAS_DILIGENT=1)
  target_link_libraries(gfx_bench PRIVATE Diligent-GraphicsEngineD3D12-shared)
  # The executable embeds this identity; the runner also verifies the DLL copy and clean checkouts.
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/competitors/diligent-build.lock.json" diligent_lock)
  set(diligent_identity "")
  foreach(dep diligent_core diligent_xxhash diligent_abseil diligent_nvapi)
    string(JSON dep_dir GET "${diligent_lock}" dependencies ${dep} directory)
    string(JSON dep_sha GET "${diligent_lock}" dependencies ${dep} commit)
    execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}/../${dep_dir}" rev-parse HEAD
      OUTPUT_VARIABLE actual_sha OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT actual_sha STREQUAL dep_sha)
      message(FATAL_ERROR "Unpinned Diligent build dependency: ${dep}")
    endif()
    string(APPEND diligent_identity "${dep}|${dep_sha}\n")
  endforeach()
  file(GLOB_RECURSE bench_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
  list(APPEND bench_sources "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
  list(SORT bench_sources)
  foreach(source IN LISTS bench_sources)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
    file(SHA256 "${source}" digest)
    file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}/.." "${source}")
    string(APPEND diligent_identity "${relative}|${digest}\n")
  endforeach()
  string(SHA256 diligent_fingerprint "${diligent_identity}")
  target_compile_definitions(gfx_bench PRIVATE GFX_DILIGENT_BUILD_SHA256="${diligent_fingerprint}")
  add_custom_command(TARGET gfx_bench POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Diligent-GraphicsEngineD3D12-shared>" "$<TARGET_FILE_DIR:gfx_bench>")
endif()
if(GFX_BENCH_WITH_SDK)
  set(GFX_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GFX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/sdk")
  target_sources(gfx_bench PRIVATE competitors/GraphicsSDK.cpp)
  target_compile_definitions(gfx_bench PRIVATE GFX_BENCH_HAS_SDK=1)
  target_link_libraries(gfx_bench PRIVATE GraphicsSDK::Core)
endif()
target_include_directories(gfx_bench PRIVATE "${LLGL_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(gfx_bench PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)

# Embed the own adapter/harness identity, including the public upload path.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/competitors/sources.lock.json" common_lock)
string(JSON llgl_pin GET "${common_lock}" llgl commit)
set(llgl_identity "llgl|${llgl_pin}\n")
file(GLOB_RECURSE llgl_bench_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/*.cmake")
list(APPEND llgl_bench_sources "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(SORT llgl_bench_sources)
foreach(source IN LISTS llgl_bench_sources)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
  file(SHA256 "${source}" digest)
  file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}/.." "${source}")
  string(APPEND llgl_identity "${relative}|${digest}\n")
endforeach()
string(SHA256 llgl_fingerprint "${llgl_identity}")
target_compile_definitions(gfx_bench PRIVATE GFX_LLGL_BUILD_SHA256="${llgl_fingerprint}")
target_compile_options(gfx_bench PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(gfx_bench PRIVATE "${LLGL_LIBRARY}" SDL3::SDL3-static d3d12 dxgi d3dcompiler advapi32)
add_custom_command(TARGET gfx_bench POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${LLGL_BINARY_DIR}/build/Release/LLGL.dll" "$<TARGET_FILE_DIR:gfx_bench>"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${LLGL_BINARY_DIR}/build/Release/LLGL_Direct3D12.dll" "$<TARGET_FILE_DIR:gfx_bench>")

enable_testing()
add_executable(gpu_timestamp_contracts tests/GpuTimestampTests.cpp)
target_include_directories(gpu_timestamp_contracts PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME gpu_timestamp_contracts COMMAND gpu_timestamp_contracts)
add_executable(telemetry_contracts tests/TelemetryTests.cpp)
target_include_directories(telemetry_contracts PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME telemetry_contracts COMMAND telemetry_contracts)
foreach(sdk llgl sdl graphicssdk)
  if(sdk STREQUAL "graphicssdk" AND NOT GFX_BENCH_WITH_SDK)
    continue()
  endif()
  foreach(scenario buffers draw upload1080 pipeline)
    add_test(NAME ${sdk}_telemetry_${scenario} COMMAND gfx_bench ${sdk} ${scenario} 1 1 3 1 --telemetry)
    set_tests_properties(${sdk}_telemetry_${scenario} PROPERTIES RUN_SERIAL TRUE TIMEOUT 120)
  endforeach()
endforeach()
add_executable(llgl_upload_contracts tests/LLGLUploadTests.cpp competitors/LLGL.cpp common/Environment.cpp)
target_include_directories(llgl_upload_contracts PRIVATE "${LLGL_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(llgl_upload_contracts PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN GFX_LLGL_BUILD_SHA256="${llgl_fingerprint}")
target_compile_options(llgl_upload_contracts PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(llgl_upload_contracts PRIVATE "${LLGL_LIBRARY}" d3d12 dxgi advapi32)
add_test(NAME llgl_upload_contracts COMMAND llgl_upload_contracts)
set_tests_properties(llgl_upload_contracts PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
foreach(scenario upload1080 upload4k upload4k16f)
  add_test(NAME llgl_${scenario}_repeated COMMAND gfx_bench llgl ${scenario} 1 2 8 1)
  set_tests_properties(llgl_${scenario}_repeated PROPERTIES RUN_SERIAL TRUE TIMEOUT 120)
endforeach()
set(timestamp_sdks llgl)
if(GFX_BENCH_WITH_SDK)
  list(APPEND timestamp_sdks graphicssdk)
endif()
if(GFX_BENCH_WITH_FORGE)
  list(APPEND timestamp_sdks the_forge)
endif()
if(GFX_BENCH_WITH_DILIGENT)
  list(APPEND timestamp_sdks diligent_core)
endif()
foreach(sdk IN LISTS timestamp_sdks)
  foreach(scenario draw_gpu instanced_gpu)
    add_test(NAME ${sdk}_${scenario} COMMAND gfx_bench ${sdk} ${scenario} 10000 2 5 1)
    set_tests_properties(${sdk}_${scenario} PROPERTIES RUN_SERIAL TRUE TIMEOUT 120)
  endforeach()
endforeach()
