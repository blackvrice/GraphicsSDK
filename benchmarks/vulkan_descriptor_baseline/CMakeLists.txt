cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKDescriptorBaseline LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows MSVC is required for this baseline")
endif()
set(GFX_INSTALLED_SDK "" CACHE PATH "Installed Vulkan SDK")
find_package(GraphicsSDK CONFIG REQUIRED PATHS "${GFX_INSTALLED_SDK}" NO_DEFAULT_PATH)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(vulkan_headers "${root}/.deps/Vulkan-Headers/include")
add_executable(vulkan_descriptor_baseline main.cpp "${root}/benchmarks/common/Environment.cpp")
target_link_libraries(vulkan_descriptor_baseline PRIVATE GraphicsSDK::Core dxgi advapi32)
target_include_directories(vulkan_descriptor_baseline PRIVATE "${root}/benchmarks/common"
    "${root}/benchmarks/vulkan_upload" "${root}/tests" "${vulkan_headers}")
add_library(VkLayer_gfx_descriptor_baseline SHARED Layer.cpp)
target_include_directories(VkLayer_gfx_descriptor_baseline SYSTEM PRIVATE "${vulkan_headers}")
foreach(target vulkan_descriptor_baseline VkLayer_gfx_descriptor_baseline)
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/VkLayer_gfx_descriptor_baseline.json" CONTENT
    [=[{"file_format_version":"1.2.0","layer":{"name":"VK_LAYER_GFX_descriptor_baseline","type":"GLOBAL","library_path":"$<TARGET_FILE:VkLayer_gfx_descriptor_baseline>","api_version":"1.1.0","implementation_version":"1","functions":{"vkNegotiateLoaderLayerInterfaceVersion":"gfxDescriptorNegotiate"},"description":"GraphicsSDK descriptor lifecycle observer"}}]=])
