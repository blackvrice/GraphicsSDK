cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKMetalAvailability LANGUAGES OBJCXX)
if(NOT APPLE OR NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  message(FATAL_ERROR "Native macOS required")
endif()
set(CMAKE_OBJCXX_STANDARD 23)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_EXTENSIONS OFF)
find_library(metal_framework Metal REQUIRED)
find_library(foundation_framework Foundation REQUIRED)
add_executable(metal_availability main.mm)
target_compile_options(metal_availability PRIVATE -fobjc-arc -Wall -Wextra -Wpedantic)
target_link_libraries(metal_availability PRIVATE "${metal_framework}" "${foundation_framework}")
