cmake_minimum_required(VERSION 3.25)
project(MetalComparison LANGUAGES C CXX OBJC OBJCXX)

find_package(GraphicsSDK CONFIG REQUIRED)
find_library(METAL Metal REQUIRED)
find_library(FOUNDATION Foundation REQUIRED)

set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
foreach(option SDL_TESTS SDL_TEST_LIBRARY SDL_EXAMPLES SDL_INSTALL SDL_AUDIO SDL_VULKAN SDL_OPENGL SDL_OPENGLES)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(option SDL_VIDEO SDL_GPU SDL_METAL SDL_OFFSCREEN)
  set(${option} ON CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${SDL_SOURCE}" sdl EXCLUDE_FROM_ALL)

foreach(option LLGL_BUILD_RENDERER_NULL LLGL_BUILD_RENDERER_OPENGL LLGL_BUILD_RENDERER_VULKAN LLGL_BUILD_TESTS LLGL_BUILD_EXAMPLES)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLGL_BUILD_STATIC_LIB ON CACHE BOOL "" FORCE)
set(LLGL_BUILD_RENDERER_METAL ON CACHE BOOL "" FORCE)
add_subdirectory("${LLGL_SOURCE}" llgl EXCLUDE_FROM_ALL)

foreach(adapter sdk sdl llgl)
  add_executable(metal_${adapter} Compare.mm)
  set_target_properties(metal_${adapter} PROPERTIES
    OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED YES OBJCXX_EXTENSIONS OFF)
  target_compile_definitions(metal_${adapter} PRIVATE COMPARE_${adapter})
  target_link_libraries(metal_${adapter} PRIVATE ${METAL} ${FOUNDATION})
endforeach()
target_link_libraries(metal_sdk PRIVATE GraphicsSDK::Core)
target_link_libraries(metal_sdl PRIVATE SDL3::SDL3-static)
target_link_libraries(metal_llgl PRIVATE LLGL_Metal LLGL)
