cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKLLGLVulkanUpload LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This audited consumer requires Windows x64 MSVC")
endif()
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../.." root)
set(LLGL_PROBE_BUILD "${root}/out/phase3-llgl-vulkan-20261028-2" CACHE PATH "Audited CMP-02 dependency build")
set(GFX_LLGL_PROFILE upstream CACHE STRING "Explicit upstream or transfer-src-candidate profile")
set(LLGL_DLL_DIRECTORY "${LLGL_PROBE_BUILD}/build/Release" CACHE PATH "Explicit runtime DLL source directory")
if(NOT GFX_LLGL_PROFILE STREQUAL "upstream" AND NOT GFX_LLGL_PROFILE STREQUAL "transfer-src-candidate")
  message(FATAL_ERROR "Unknown LLGL comparison profile")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
file(READ "${root}/benchmarks/competitors/sources.lock.json" lock)
string(JSON pin GET "${lock}" llgl commit)
file(GLOB own CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
list(APPEND own "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${root}/benchmarks/competitors/sources.lock.json"
  "${root}/benchmarks/vulkan_upload/main.cpp" "${root}/benchmarks/phase3_comparison/upload.json")
list(SORT own)
set(identity "llgl|${pin}\ndependency_profile|${GFX_LLGL_PROFILE}\n")
foreach(p IN LISTS own)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${p}")
  file(SHA256 "${p}" digest)
  file(RELATIVE_PATH name "${root}" "${p}")
  string(APPEND identity "${name}|${digest}\n")
endforeach()
string(SHA256 fingerprint "${identity}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source-identity.txt" "${identity}")
add_executable(llgl_vulkan_upload main.cpp)
target_compile_features(llgl_vulkan_upload PRIVATE cxx_std_23)
target_compile_options(llgl_vulkan_upload PRIVATE /W4 /permissive- /EHsc /utf-8)
target_compile_definitions(llgl_vulkan_upload PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN
  GFX_LLGL_PIN="${pin}" GFX_LLGL_UPLOAD_FINGERPRINT="${fingerprint}" GFX_LLGL_PROFILE="${GFX_LLGL_PROFILE}")
target_include_directories(llgl_vulkan_upload PRIVATE "${root}/.deps/LLGL/include" "${root}/.deps/Vulkan-Headers/include")
target_link_libraries(llgl_vulkan_upload PRIVATE "${LLGL_PROBE_BUILD}/build/llgl/build/Release/LLGL.lib" "${LLGL_PROBE_BUILD}/vulkan-1.lib" bcrypt)
add_custom_command(TARGET llgl_vulkan_upload POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${LLGL_DLL_DIRECTORY}/LLGL.dll" "$<TARGET_FILE_DIR:llgl_vulkan_upload>"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${LLGL_DLL_DIRECTORY}/LLGL_Vulkan.dll" "$<TARGET_FILE_DIR:llgl_vulkan_upload>")
