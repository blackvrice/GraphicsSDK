cmake_minimum_required(VERSION 3.25)
project(LLGLTransferSrcCandidate LANGUAGES C CXX)
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../.." root)
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${GFX_LLGL_OVERLAY}")
  message(FATAL_ERROR "Windows MSVC and explicit reviewed overlay required")
endif()
set(LLGL_BUILD_RENDERER_VULKAN ON CACHE BOOL "" FORCE)
foreach(name NULL OPENGL DIRECT3D11 DIRECT3D12 METAL WEBGL)
  set(LLGL_BUILD_RENDERER_${name} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(name LLGL_BUILD_STATIC_LIB LLGL_BUILD_EXAMPLES LLGL_BUILD_TESTS LLGL_BUILD_WRAPPER_C99 LLGL_BUILD_XR_OPENXR LLGL_ENABLE_DEBUG_LAYER LLGL_VK_ENABLE_SPIRV_REFLECT LLGL_VK_ENABLE_GPU_ASSISTED_VALIDATION)
  set(${name} OFF CACHE BOOL "" FORCE)
endforeach()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
add_subdirectory("${root}/.deps/LLGL" "${CMAKE_CURRENT_BINARY_DIR}/llgl" EXCLUDE_FROM_ALL)
get_target_property(sources LLGL_Vulkan SOURCES)
set(original "${root}/.deps/LLGL/sources/Renderer/Vulkan/Command/VKCommandContext.cpp")
list(FIND sources "${original}" index)
if(index EQUAL -1)
  message(FATAL_ERROR "Pinned Vulkan command context not found")
endif()
list(REMOVE_ITEM sources "${original}")
set_property(TARGET LLGL_Vulkan PROPERTY SOURCES "${sources}")
target_sources(LLGL_Vulkan PRIVATE "${GFX_LLGL_OVERLAY}")
target_include_directories(LLGL_Vulkan PRIVATE "${root}/.deps/LLGL/sources/Renderer/Vulkan/Command")
add_custom_target(llgl_transfer_src_candidate ALL DEPENDS LLGL_Vulkan)
