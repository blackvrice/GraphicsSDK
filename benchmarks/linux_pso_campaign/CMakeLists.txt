cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKLinuxPSOCampaign LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Native Linux x86_64 required")
endif()
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../.." root)
set(GFX_CAMPAIGN_MODE "software-diagnostic" CACHE STRING "Qualified native device kind")
if(NOT EXISTS "${LLGL_PROBE_BUILD}/lib/libLLGL.so" OR NOT EXISTS "${LLGL_PROBE_BUILD}/lib/libLLGL_Vulkan.so")
  message(FATAL_ERROR "Qualified approved LLGL shared-library build required")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OPENSSL_USE_STATIC_LIBS OFF)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_UNITY_BUILD OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${root}/tools/port_linux_pso_campaign.py"
  --output "${CMAKE_CURRENT_BINARY_DIR}" --mode "${GFX_CAMPAIGN_MODE}" COMMAND_ERROR_IS_FATAL ANY)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/adaptation.json" manifest)
string(JSON fingerprint GET "${manifest}" source_fingerprint)
file(READ "${root}/benchmarks/competitors/sources.lock.json" lock)
string(JSON pin GET "${lock}" llgl commit)
string(JSON count LENGTH "${manifest}" inputs)
math(EXPR last "${count}-1")
foreach(i RANGE 0 ${last})
  string(JSON name GET "${manifest}" inputs ${i} path)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${root}/${name}")
endforeach()
add_library(upload_observer SHARED "${CMAKE_CURRENT_BINARY_DIR}/Layer.cpp")
set_target_properties(upload_observer PROPERTIES OUTPUT_NAME VkLayer_gfx_llgl_draw)
target_compile_features(upload_observer PRIVATE cxx_std_23)
target_include_directories(upload_observer PRIVATE "${root}/benchmarks/llgl_vulkan_draw" "${root}/.deps/Vulkan-Headers/include")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/VkLayer_gfx_llgl_draw.json" CONTENT "{\n\"file_format_version\":\"1.2.0\",\"layer\":{\"name\":\"VK_LAYER_GFX_llgl_draw\",\"type\":\"GLOBAL\",\"library_path\":\"./libVkLayer_gfx_llgl_draw.so\",\"api_version\":\"1.3.290\",\"implementation_version\":\"1\",\"description\":\"Own Linux pass-through draw observer\",\"functions\":{\"vkNegotiateLoaderLayerInterfaceVersion\":\"gfxDrawNegotiate\"}}}\n")
add_executable(linux_pso_compare "${CMAKE_CURRENT_BINARY_DIR}/main.cpp")
add_dependencies(linux_pso_compare upload_observer)
target_compile_features(linux_pso_compare PRIVATE cxx_std_23)
target_compile_options(linux_pso_compare PRIVATE -Wall -Wextra -Wpedantic)
file(SHA256 "${root}/tests/fixtures/SpirvFixtures.hpp" shader_sha)
target_compile_definitions(linux_pso_compare PRIVATE GFX_PSO_SHADER_SHA="${shader_sha}" GFX_LLGL_PIN="${pin}" GFX_LLGL_PSO_COMPARE_FINGERPRINT="${fingerprint}")
target_include_directories(linux_pso_compare PRIVATE "${root}/benchmarks/common" "${root}/benchmarks/llgl_vulkan_draw" "${root}/benchmarks/vulkan_pso" "${root}/tests/fixtures")
target_include_directories(linux_pso_compare SYSTEM PRIVATE "${root}/.deps/LLGL/include" "${root}/.deps/Vulkan-Headers/include")
target_link_libraries(linux_pso_compare PRIVATE "${LLGL_PROBE_BUILD}/lib/libLLGL.so" "${Vulkan_LIBRARY}" OpenSSL::Crypto ${CMAKE_DL_LIBS})
