cmake_minimum_required(VERSION 3.25)
project(VulkanUploadCPUProfile LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This CPU API interval profile is verified on Windows MSVC only")
endif()
set(GFX_INSTALLED_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../out/vulkan-presentation-install" CACHE PATH "Installed SDK to profile")
find_package(GraphicsSDK CONFIG REQUIRED PATHS "${GFX_INSTALLED_SDK}" NO_DEFAULT_PATH)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(inputs "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${root}/benchmarks/vulkan_upload/Identity.hpp" "${root}/benchmarks/common/Adapter.hpp" "${root}/benchmarks/common/Environment.cpp")
list(SORT inputs)
set(identity "")
foreach(p IN LISTS inputs)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${p}")
  file(SHA256 "${p}" sha)
  file(RELATIVE_PATH relative "${root}" "${p}")
  string(APPEND identity "${relative}|${sha}\n")
endforeach()
string(SHA256 fingerprint "${identity}")
add_executable(vulkan_upload_profile main.cpp ../common/Environment.cpp)
target_include_directories(vulkan_upload_profile PRIVATE "${root}/.deps/Vulkan-Headers/include")
target_compile_features(vulkan_upload_profile PRIVATE cxx_std_23)
target_compile_definitions(vulkan_upload_profile PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN GFX_PROFILE_SHA="${fingerprint}")
target_compile_options(vulkan_upload_profile PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(vulkan_upload_profile PRIVATE GraphicsSDK::Core dxgi advapi32)
