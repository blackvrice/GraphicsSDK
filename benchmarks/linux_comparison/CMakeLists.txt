cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKLinuxComparisonProbe LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Native Linux probe required")
endif()
find_package(GraphicsSDK CONFIG REQUIRED)
find_package(SDL3 CONFIG REQUIRED)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
file(READ "${root}/benchmarks/competitors/sources.lock.json" lock)
string(JSON pin GET "${lock}" sdl commit)
set(own "${CMAKE_CURRENT_SOURCE_DIR}/Probe.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/Selection.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${root}/benchmarks/common/LinuxEnvironment.hpp" "${root}/benchmarks/competitors/sources.lock.json")
list(SORT own)
set(identity "")
foreach(p IN LISTS own)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${p}")
  file(SHA256 "${p}" sha)
  file(RELATIVE_PATH relative "${root}" "${p}")
  string(APPEND identity "${relative}|${sha}\n")
endforeach()
string(SHA256 fingerprint "${identity}")
add_executable(linux_comparison Probe.cpp)
target_compile_features(linux_comparison PRIVATE cxx_std_23)
target_include_directories(linux_comparison SYSTEM PRIVATE "${root}/.deps/Vulkan-Headers/include")
target_compile_options(linux_comparison PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(linux_comparison PRIVATE GFX_LINUX_COMPARISON_FINGERPRINT="${fingerprint}" GFX_SDL_PIN="${pin}")
target_link_libraries(linux_comparison PRIVATE GraphicsSDK::Core SDL3::SDL3-static ${CMAKE_DL_LIBS})
add_executable(linux_comparison_selection Selection.cpp)
target_compile_features(linux_comparison_selection PRIVATE cxx_std_23)
target_include_directories(linux_comparison_selection SYSTEM PRIVATE "${root}/.deps/Vulkan-Headers/include")
target_link_libraries(linux_comparison_selection PRIVATE ${CMAKE_DL_LIBS})
