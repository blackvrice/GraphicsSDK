cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKDrawTimestamp LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Dedicated Windows x64 MSVC timestamp layer profile required")
endif()
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../.." root)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
file(GLOB own CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
list(APPEND own "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(SORT own)
set(identity "profile|native_render_pass_gpu_timestamp_v1\n")
foreach(p IN LISTS own)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${p}")
  file(SHA256 "${p}" digest)
  file(RELATIVE_PATH name "${root}" "${p}")
  string(APPEND identity "${name}|${digest}\n")
endforeach()
string(SHA256 fingerprint "${identity}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source-identity.txt" "${identity}")
add_library(draw_timestamp SHARED Layer.cpp)
set_target_properties(draw_timestamp PROPERTIES OUTPUT_NAME VkLayer_gfx_draw_timestamp)
target_include_directories(draw_timestamp PRIVATE "${root}/.deps/Vulkan-Headers/include")
target_compile_features(draw_timestamp PRIVATE cxx_std_23)
target_compile_options(draw_timestamp PRIVATE /W4 /permissive- /EHsc /utf-8)
target_compile_definitions(draw_timestamp PRIVATE GFX_TIMESTAMP_FINGERPRINT="${fingerprint}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/VkLayer_gfx_draw_timestamp.json" CONTENT "{\n\"file_format_version\":\"1.2.0\",\"layer\":{\"name\":\"VK_LAYER_GFX_draw_timestamp\",\"type\":\"GLOBAL\",\"library_path\":\"$<TARGET_FILE:draw_timestamp>\",\"api_version\":\"1.3.290\",\"implementation_version\":\"1\",\"description\":\"Own dedicated render pass timestamp measurement\",\"functions\":{\"vkNegotiateLoaderLayerInterfaceVersion\":\"gfxTimestampNegotiate\"}}}\n")
