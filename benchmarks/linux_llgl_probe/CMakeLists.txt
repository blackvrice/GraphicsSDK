cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKLinuxLLGLProbe LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Native Linux x86_64 required")
endif()
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../.." root)
if(NOT EXISTS "${GFX_LLGL_OVERLAY}")
  message(FATAL_ERROR "Explicit approved transfer-src overlay required")
endif()
file(READ "${root}/benchmarks/competitors/sources.lock.json" lock)
string(JSON pin GET "${lock}" llgl commit)
execute_process(COMMAND git --no-optional-locks -C "${root}/.deps/LLGL" rev-parse HEAD
  OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual STREQUAL pin)
  message(FATAL_ERROR "LLGL pin differs")
endif()
set(LLGL_BUILD_RENDERER_VULKAN ON CACHE BOOL "" FORCE)
foreach(name NULL OPENGL DIRECT3D11 DIRECT3D12 METAL WEBGL)
  set(LLGL_BUILD_RENDERER_${name} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(name LLGL_BUILD_STATIC_LIB LLGL_BUILD_EXAMPLES LLGL_BUILD_TESTS LLGL_BUILD_WRAPPER_C99
    LLGL_BUILD_XR_OPENXR LLGL_ENABLE_DEBUG_LAYER LLGL_VK_ENABLE_SPIRV_REFLECT
    LLGL_VK_ENABLE_GPU_ASSISTED_VALIDATION LLGL_LINUX_ENABLE_WAYLAND)
  set(${name} OFF CACHE BOOL "" FORCE)
endforeach()
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_UNITY_BUILD OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
set(CMAKE_C_FLAGS_RELEASE "-O2 -DNDEBUG")
set(LLGL_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/lib" CACHE PATH "" FORCE)
add_subdirectory("${root}/.deps/LLGL" "${CMAKE_CURRENT_BINARY_DIR}/llgl" EXCLUDE_FROM_ALL)
get_target_property(sources LLGL_Vulkan SOURCES)
set(original "${root}/.deps/LLGL/sources/Renderer/Vulkan/Command/VKCommandContext.cpp")
list(FIND sources "${original}" index)
if(index EQUAL -1)
  message(FATAL_ERROR "Pinned Vulkan command context not found")
endif()
list(REMOVE_ITEM sources "${original}")
set_property(TARGET LLGL_Vulkan PROPERTY SOURCES "${sources}")
target_sources(LLGL_Vulkan PRIVATE "${GFX_LLGL_OVERLAY}")
target_include_directories(LLGL_Vulkan PRIVATE "${root}/.deps/LLGL/sources/Renderer/Vulkan/Command")
set(own "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt"
    "${root}/benchmarks/common/LinuxEnvironment.hpp" "${root}/benchmarks/competitors/sources.lock.json")
list(SORT own)
set(identity "linux-llgl|${pin}|llgl-fixed-pin-transfer-src-v1\n")
foreach(p IN LISTS own)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${p}")
  file(SHA256 "${p}" sha)
  file(RELATIVE_PATH relative "${root}" "${p}")
  string(APPEND identity "${relative}|${sha}\n")
endforeach()
file(SHA256 "${GFX_LLGL_OVERLAY}" overlay_sha)
string(APPEND identity "overlay|${overlay_sha}\n")
string(SHA256 fingerprint "${identity}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/probe-identity.txt" "${identity}")
add_executable(linux_llgl_probe main.cpp)
add_dependencies(linux_llgl_probe LLGL_Vulkan)
target_compile_features(linux_llgl_probe PRIVATE cxx_std_23)
target_compile_options(linux_llgl_probe PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(linux_llgl_probe PRIVATE GFX_LLGL_PIN="${pin}" GFX_LLGL_FINGERPRINT="${fingerprint}")
target_include_directories(linux_llgl_probe SYSTEM PRIVATE "${root}/.deps/LLGL/include" "${Vulkan_INCLUDE_DIR}")
target_link_libraries(linux_llgl_probe PRIVATE LLGL "${Vulkan_LIBRARY}" ${CMAKE_DL_LIBS})
