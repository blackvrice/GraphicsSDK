cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKVulkanFrameSlots LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows MSVC is required for this baseline")
endif()
set(GFX_INSTALLED_SDK "" CACHE PATH "Installed Vulkan SDK")
find_package(GraphicsSDK CONFIG REQUIRED PATHS "${GFX_INSTALLED_SDK}" NO_DEFAULT_PATH)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_executable(vulkan_frame_slots main.cpp "${root}/benchmarks/common/Environment.cpp")
target_link_libraries(vulkan_frame_slots PRIVATE GraphicsSDK::Core dxgi advapi32)
target_compile_features(vulkan_frame_slots PRIVATE cxx_std_23)
target_compile_definitions(vulkan_frame_slots PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(vulkan_frame_slots PRIVATE /W4 /permissive- /EHsc /utf-8)
target_include_directories(vulkan_frame_slots PRIVATE
    "${root}/benchmarks/common" "${root}/benchmarks/vulkan_upload" "${root}/.deps/Vulkan-Headers/include")
