cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKLinuxLLGLUpload LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Native Linux x86_64 required")
endif()
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../.." root)
if(NOT EXISTS "${LLGL_PROBE_BUILD}/lib/libLLGL.so" OR NOT EXISTS "${LLGL_PROBE_BUILD}/lib/libLLGL_Vulkan.so")
  message(FATAL_ERROR "Qualified approved LLGL shared-library build required")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OPENSSL_USE_STATIC_LIBS OFF)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_UNITY_BUILD OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${root}/tools/port_linux_llgl_upload.py"
  --output "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/adaptation.json" manifest)
string(JSON fingerprint GET "${manifest}" source_fingerprint)
file(READ "${root}/benchmarks/competitors/sources.lock.json" lock)
string(JSON pin GET "${lock}" llgl commit)
foreach(p main.cpp Context.hpp Payload.hpp)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${root}/benchmarks/llgl_vulkan_upload/${p}")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${root}/tools/port_linux_llgl_upload.py"
  "${root}/benchmarks/common/LinuxEnvironment.hpp" "${root}/benchmarks/competitors/sources.lock.json"
  "${root}/benchmarks/phase3_comparison/upload.json")
add_executable(linux_llgl_upload "${CMAKE_CURRENT_BINARY_DIR}/main.cpp")
target_compile_features(linux_llgl_upload PRIVATE cxx_std_23)
target_compile_options(linux_llgl_upload PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(linux_llgl_upload PRIVATE GFX_LLGL_PIN="${pin}"
  GFX_LLGL_UPLOAD_FINGERPRINT="${fingerprint}" GFX_LLGL_PROFILE="llgl-fixed-pin-transfer-src-v1")
target_include_directories(linux_llgl_upload PRIVATE "${root}/benchmarks/common" "${root}/benchmarks/llgl_vulkan_upload")
target_include_directories(linux_llgl_upload SYSTEM PRIVATE "${root}/.deps/LLGL/include" "${root}/.deps/Vulkan-Headers/include")
target_link_libraries(linux_llgl_upload PRIVATE "${LLGL_PROBE_BUILD}/lib/libLLGL.so" "${Vulkan_LIBRARY}" OpenSSL::Crypto ${CMAKE_DL_LIBS})
