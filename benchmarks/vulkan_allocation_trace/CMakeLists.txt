cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKVulkanAllocationTrace LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This measured profile requires Windows x64 MSVC; Linux remains unverified")
endif()
set(GFX_INSTALLED_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../out/vulkan-presentation-install" CACHE PATH "Frozen installed Vulkan SDK")
set(SDL_BINARY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../out/sdl" CACHE PATH "Reviewed SDL static build")
find_package(GraphicsSDK CONFIG REQUIRED PATHS "${GFX_INSTALLED_SDK}" NO_DEFAULT_PATH)
find_package(SDL3 CONFIG REQUIRED PATHS "${SDL_BINARY_DIR}" NO_DEFAULT_PATH)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(READ "${root}/benchmarks/competitors/sources.lock.json" lock)
string(JSON sdl_pin GET "${lock}" sdl commit)
execute_process(COMMAND git -c gc.auto=0 -c maintenance.auto=false -C "${root}/.deps/SDL" rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual STREQUAL sdl_pin)
  message(FATAL_ERROR "SDL checkout differs from reviewed pin")
endif()
file(GLOB own CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
list(APPEND own "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${root}/benchmarks/vulkan_upload/Identity.hpp" "${root}/benchmarks/vulkan_allocation/Payload.hpp" "${root}/licenses/vulkan-allocation-trace-policy.json" "${root}/benchmarks/common/Adapter.hpp" "${root}/benchmarks/common/Environment.cpp" "${root}/benchmarks/competitors/sources.lock.json")
list(SORT own)
set(identity "")
foreach(p IN LISTS own)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${p}")
  file(SHA256 "${p}" sha)
  file(RELATIVE_PATH relative "${root}" "${p}")
  string(APPEND identity "${relative}|${sha}\n")
endforeach()
string(SHA256 fingerprint "${identity}")
add_executable(vulkan_allocation_trace main.cpp ../common/Environment.cpp)
target_include_directories(vulkan_allocation_trace PRIVATE "${root}/.deps/Vulkan-Headers/include")
target_compile_features(vulkan_allocation_trace PRIVATE cxx_std_23)
target_compile_definitions(vulkan_allocation_trace PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN GFX_TRACE_FINGERPRINT="${fingerprint}" GFX_SDL_PIN="${sdl_pin}")
target_compile_options(vulkan_allocation_trace PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(vulkan_allocation_trace PRIVATE GraphicsSDK::Core SDL3::SDL3-static dxgi advapi32)
enable_testing()
add_executable(vulkan_allocation_trace_reference "${root}/tests/VulkanAllocationTraceTests.cpp")
target_compile_features(vulkan_allocation_trace_reference PRIVATE cxx_std_23)
target_compile_options(vulkan_allocation_trace_reference PRIVATE /W4 /permissive- /EHsc /utf-8)
add_test(NAME trace_ledger_reference COMMAND vulkan_allocation_trace_reference)
add_library(VkLayer_gfx_allocation SHARED Layer.cpp)
target_include_directories(VkLayer_gfx_allocation PRIVATE "${root}/.deps/Vulkan-Headers/include")
target_compile_features(VkLayer_gfx_allocation PRIVATE cxx_std_23)
target_compile_definitions(VkLayer_gfx_allocation PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN GFX_TRACE_FINGERPRINT="${fingerprint}")
target_compile_options(VkLayer_gfx_allocation PRIVATE /W4 /permissive- /EHsc /utf-8)
add_executable(vulkan_allocation_trace_interface "${root}/tests/VulkanAllocationTraceInterfaceTests.cpp")
target_include_directories(vulkan_allocation_trace_interface PRIVATE "${root}/.deps/Vulkan-Headers/include")
target_compile_features(vulkan_allocation_trace_interface PRIVATE cxx_std_23)
target_compile_definitions(vulkan_allocation_trace_interface PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(vulkan_allocation_trace_interface PRIVATE /W4 /permissive- /EHsc /utf-8)
add_test(NAME trace_layer_interface COMMAND vulkan_allocation_trace_interface "$<TARGET_FILE:VkLayer_gfx_allocation>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/VkLayer_gfx_allocation.json" CONTENT "{\"file_format_version\":\"1.2.0\",\"layer\":{\"name\":\"VK_LAYER_GFX_allocation_trace\",\"type\":\"GLOBAL\",\"library_path\":\"./VkLayer_gfx_allocation.dll\",\"api_version\":\"1.1.0\",\"implementation_version\":\"1\",\"functions\":{\"vkNegotiateLoaderLayerInterfaceVersion\":\"gfxNegotiateLayer\"},\"description\":\"GraphicsSDK bounded native allocation diagnostics\"}}\n")
