cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKLinuxWorkloads LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Native Linux workload build required")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GFX_INSTALLED_SDK "" CACHE PATH "Qualified installed SDK root")
set(SDL_BINARY_DIR "" CACHE PATH "Qualified pinned static SDL build root")
find_package(GraphicsSDK CONFIG REQUIRED PATHS "${GFX_INSTALLED_SDK}" NO_DEFAULT_PATH)
find_package(SDL3 CONFIG REQUIRED PATHS "${SDL_BINARY_DIR}" NO_DEFAULT_PATH)
get_filename_component(root "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
file(READ "${root}/benchmarks/competitors/sources.lock.json" lock)
string(JSON pin GET "${lock}" sdl commit)
file(SHA256 "${root}/tests/fixtures/SpirvFixtures.hpp" shader_sha)
foreach(profile upload draw pso allocation)
  file(GLOB original CONFIGURE_DEPENDS "${root}/benchmarks/vulkan_${profile}/*.cpp" "${root}/benchmarks/vulkan_${profile}/*.hpp")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${original} "${root}/tools/port_linux_benchmarks.py" "${root}/benchmarks/common/LinuxEnvironment.hpp" "${root}/benchmarks/common/Adapter.hpp" "${root}/benchmarks/competitors/sources.lock.json" "${root}/tests/fixtures/SpirvFixtures.hpp")
  file(GLOB fixture_assembly CONFIGURE_DEPENDS "${root}/tests/fixtures/spirv/*.spvasm")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Platform.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/Environment.cpp" ${fixture_assembly})
  set(generated "${CMAKE_CURRENT_BINARY_DIR}/${profile}.cpp")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${root}/tools/port_linux_benchmarks.py" --profile "${profile}" --output "${generated}" COMMAND_ERROR_IS_FATAL ANY)
  file(READ "${CMAKE_CURRENT_BINARY_DIR}/${profile}.json" manifest)
  string(JSON fingerprint GET "${manifest}" source_fingerprint)
  string(TOUPPER "${profile}" upper)
  add_executable(linux_vulkan_${profile} "${generated}" Environment.cpp)
  target_compile_features(linux_vulkan_${profile} PRIVATE cxx_std_23)
  target_include_directories(linux_vulkan_${profile} PRIVATE "${root}/benchmarks/vulkan_${profile}")
  target_include_directories(linux_vulkan_${profile} SYSTEM PRIVATE "${root}/.deps/Vulkan-Headers/include")
  target_compile_options(linux_vulkan_${profile} PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_definitions(linux_vulkan_${profile} PRIVATE GFX_VK_${upper}_FINGERPRINT="${fingerprint}" GFX_SDL_PIN="${pin}" GFX_${upper}_SHADER_HEADER_SHA="${shader_sha}")
  target_link_libraries(linux_vulkan_${profile} PRIVATE GraphicsSDK::Core SDL3::SDL3-static ${CMAKE_DL_LIBS})
endforeach()
