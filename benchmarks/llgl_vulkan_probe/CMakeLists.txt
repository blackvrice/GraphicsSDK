cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKLLGLVulkanProbe LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This input-audited probe requires Windows x64 MSVC")
endif()
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../.." root)
set(LLGL_SOURCE_DIR "${root}/.deps/LLGL" CACHE PATH "Pinned LLGL source")
file(READ "${root}/benchmarks/competitors/sources.lock.json" lock)
string(JSON pin GET "${lock}" llgl commit)
execute_process(COMMAND git --no-optional-locks -C "${LLGL_SOURCE_DIR}" rev-parse HEAD
  OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual STREQUAL pin)
  message(FATAL_ERROR "LLGL pin drift")
endif()
set(LLGL_BUILD_RENDERER_VULKAN ON CACHE BOOL "" FORCE)
foreach(name NULL OPENGL DIRECT3D11 DIRECT3D12 METAL WEBGL)
  set(LLGL_BUILD_RENDERER_${name} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(name LLGL_BUILD_STATIC_LIB LLGL_BUILD_EXAMPLES LLGL_BUILD_TESTS
    LLGL_BUILD_WRAPPER_C99 LLGL_BUILD_XR_OPENXR LLGL_ENABLE_DEBUG_LAYER
    LLGL_VK_ENABLE_SPIRV_REFLECT LLGL_VK_ENABLE_GPU_ASSISTED_VALIDATION)
  set(${name} OFF CACHE BOOL "" FORCE)
endforeach()
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_subdirectory("${LLGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/llgl" EXCLUDE_FROM_ALL)
file(GLOB own CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
list(APPEND own "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${root}/benchmarks/competitors/sources.lock.json")
list(SORT own)
set(identity "llgl|${pin}\n")
foreach(p IN LISTS own)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${p}")
  file(SHA256 "${p}" sha)
  file(RELATIVE_PATH relative "${root}" "${p}")
  string(APPEND identity "${relative}|${sha}\n")
endforeach()
string(SHA256 fingerprint "${identity}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/probe-identity.txt" "${identity}")
add_executable(llgl_vulkan_probe main.cpp)
add_dependencies(llgl_vulkan_probe LLGL_Vulkan)
target_compile_features(llgl_vulkan_probe PRIVATE cxx_std_23)
target_compile_options(llgl_vulkan_probe PRIVATE /W4 /permissive- /EHsc /utf-8)
target_compile_definitions(llgl_vulkan_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN
  GFX_LLGL_PIN="${pin}" GFX_LLGL_PROBE_FINGERPRINT="${fingerprint}")
target_include_directories(llgl_vulkan_probe PRIVATE "${LLGL_SOURCE_DIR}/include" "${Vulkan_INCLUDE_DIR}")
target_link_libraries(llgl_vulkan_probe PRIVATE LLGL "${Vulkan_LIBRARY}")
add_custom_command(TARGET llgl_vulkan_probe POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:LLGL>" "$<TARGET_FILE_DIR:llgl_vulkan_probe>"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:LLGL_Vulkan>" "$<TARGET_FILE_DIR:llgl_vulkan_probe>")
