cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKXcbSurfaceContracts LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Actual Linux required")
endif()
find_package(GraphicsSDK CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(gfx_surface_proxy SHARED Loader.cpp)
set_target_properties(gfx_surface_proxy PROPERTIES OUTPUT_NAME vulkan SOVERSION 1 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/proxy")
target_include_directories(gfx_surface_proxy SYSTEM PRIVATE "${root}/.deps/Vulkan-Headers/include")
target_include_directories(gfx_surface_proxy SYSTEM PRIVATE ${XCB_INCLUDE_DIRS})
target_link_libraries(gfx_surface_proxy PRIVATE ${CMAKE_DL_LIBS})
add_executable(gfx_xcb_surface Probe.cpp)
target_link_libraries(gfx_xcb_surface PRIVATE GraphicsSDK::Core PkgConfig::XCB ${CMAKE_DL_LIBS})
foreach(target gfx_surface_proxy gfx_xcb_surface)
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
