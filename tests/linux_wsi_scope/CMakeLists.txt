cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKLinuxWSIScope LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This native XCB scope probe requires Linux")
endif()
find_package(GraphicsSDK CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(GFX_VULKAN_HEADERS "${root}/.deps/Vulkan-Headers" CACHE PATH "Reviewed external headers")
add_executable(gfx_linux_wsi_scope Probe.cpp)
target_compile_features(gfx_linux_wsi_scope PRIVATE cxx_std_23)
target_include_directories(gfx_linux_wsi_scope SYSTEM PRIVATE "${GFX_VULKAN_HEADERS}/include")
target_link_libraries(gfx_linux_wsi_scope PRIVATE GraphicsSDK::Core PkgConfig::XCB ${CMAKE_DL_LIBS})
target_compile_options(gfx_linux_wsi_scope PRIVATE -Wall -Wextra -Wpedantic)
