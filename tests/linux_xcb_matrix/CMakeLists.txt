cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKXcbMatrix LANGUAGES CXX)
find_package(GraphicsSDK CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(gfx_matrix_proxy SHARED "${root}/tests/linux_xcb_surface/Loader.cpp")
set_target_properties(gfx_matrix_proxy PROPERTIES OUTPUT_NAME vulkan SOVERSION 1 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/proxy")
target_compile_definitions(gfx_matrix_proxy PRIVATE GFX_LIFETIME_OBSERVER=1)
target_include_directories(gfx_matrix_proxy SYSTEM PRIVATE "${root}/.deps/Vulkan-Headers/include" ${XCB_INCLUDE_DIRS})
target_link_libraries(gfx_matrix_proxy PRIVATE ${CMAKE_DL_LIBS})
add_executable(gfx_xcb_matrix Probe.cpp)
target_link_libraries(gfx_xcb_matrix PRIVATE GraphicsSDK::Core PkgConfig::XCB ${CMAKE_DL_LIBS})
add_executable(gfx_xcb_validation_controls ../VulkanValidationTests.cpp)
target_include_directories(gfx_xcb_validation_controls PRIVATE "${root}/src/backends/vulkan")
target_include_directories(gfx_xcb_validation_controls SYSTEM PRIVATE "${root}/.deps/Vulkan-Headers/include")
target_link_libraries(gfx_xcb_validation_controls PRIVATE GraphicsSDK::Core ${CMAKE_DL_LIBS})
foreach(target gfx_matrix_proxy gfx_xcb_matrix gfx_xcb_validation_controls)
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
