cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKLinuxTimestampMetadata LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Linux timestamp metadata probe requires Linux")
endif()
find_package(GraphicsSDK CONFIG REQUIRED)
set(GFX_VULKAN_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../../.deps/Vulkan-Headers" CACHE PATH "Reviewed Vulkan-Headers root")
if(NOT EXISTS "${GFX_VULKAN_HEADERS}/include/vulkan/vulkan_core.h")
  message(FATAL_ERROR "Reviewed Vulkan-Headers required")
endif()
add_executable(gfx_linux_timestamp_metadata Metadata.cpp)
target_link_libraries(gfx_linux_timestamp_metadata PRIVATE GraphicsSDK::Core ${CMAKE_DL_LIBS})
target_include_directories(gfx_linux_timestamp_metadata SYSTEM PRIVATE "${GFX_VULKAN_HEADERS}/include")
target_compile_options(gfx_linux_timestamp_metadata PRIVATE -Wall -Wextra -Wpedantic)
