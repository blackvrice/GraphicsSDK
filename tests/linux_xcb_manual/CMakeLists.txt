cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKXcbManual LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The XCB manual preparation consumer requires Linux")
endif()
find_package(GraphicsSDK CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(gfx_manual_proxy SHARED "${root}/tests/linux_xcb_surface/Loader.cpp")
set_target_properties(gfx_manual_proxy PROPERTIES OUTPUT_NAME vulkan SOVERSION 1 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/proxy")
target_include_directories(gfx_manual_proxy SYSTEM PRIVATE "${root}/.deps/Vulkan-Headers/include" ${XCB_INCLUDE_DIRS})
target_link_libraries(gfx_manual_proxy PRIVATE ${CMAKE_DL_LIBS})
add_executable(gfx_xcb_manual Probe.cpp)
target_link_libraries(gfx_xcb_manual PRIVATE GraphicsSDK::Core PkgConfig::XCB ${CMAKE_DL_LIBS})
foreach(target gfx_manual_proxy gfx_xcb_manual)
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
add_subdirectory("${root}/tests/linux_vulkan_device" identity)
