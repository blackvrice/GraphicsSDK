cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKLinuxDeviceProbe LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The Linux Vulkan device probe requires Linux")
endif()
find_package(GraphicsSDK CONFIG REQUIRED)
set(GFX_VULKAN_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../../.deps/Vulkan-Headers" CACHE PATH "Reviewed Vulkan-Headers checkout root")
if(NOT EXISTS "${GFX_VULKAN_HEADERS}/include/vulkan/vulkan_core.h")
  message(FATAL_ERROR "Reviewed Vulkan-Headers required for independent native identity queries")
endif()
add_executable(gfx_linux_vulkan_device Device.cpp)
target_link_libraries(gfx_linux_vulkan_device PRIVATE GraphicsSDK::Core ${CMAKE_DL_LIBS})
target_include_directories(gfx_linux_vulkan_device SYSTEM PRIVATE "${GFX_VULKAN_HEADERS}/include")
target_compile_options(gfx_linux_vulkan_device PRIVATE -Wall -Wextra -Wpedantic)
