cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKRecoveryObservation LANGUAGES CXX)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(GraphicsSDK CONFIG REQUIRED)
set(GFX_VULKAN_HEADERS "${root}/.deps/Vulkan-Headers" CACHE PATH "Reviewed Vulkan-Headers checkout")
add_library(VkLayer_gfx_recovery SHARED Layer.cpp)
target_include_directories(VkLayer_gfx_recovery SYSTEM PRIVATE "${GFX_VULKAN_HEADERS}/include")
add_executable(gfx_recovery_observe "${root}/tests/VulkanRecoveryTests.cpp")
target_link_libraries(gfx_recovery_observe PRIVATE GraphicsSDK::Core)
foreach(target VkLayer_gfx_recovery gfx_recovery_observe)
  target_compile_features(${target} PRIVATE cxx_std_23)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(${target} PRIVATE ${CMAKE_DL_LIBS})
  endif()
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/VkLayer_gfx_recovery.json" CONTENT "{\"file_format_version\":\"1.2.0\",\"layer\":{\"name\":\"VK_LAYER_GFX_recovery\",\"type\":\"GLOBAL\",\"library_path\":\"$<TARGET_FILE:VkLayer_gfx_recovery>\",\"api_version\":\"1.1.0\",\"implementation_version\":\"1\",\"functions\":{\"vkNegotiateLoaderLayerInterfaceVersion\":\"gfxRecoveryNegotiate\"},\"description\":\"GraphicsSDK controlled recovery contracts\"}}\n")
