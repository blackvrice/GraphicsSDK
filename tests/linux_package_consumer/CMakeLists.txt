cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKExternalLinuxConsumer LANGUAGES CXX)
find_package(GraphicsSDK 0.0.1 EXACT CONFIG REQUIRED)
add_executable(cpu_consumer consumer/main.cpp)
target_link_libraries(cpu_consumer PRIVATE GraphicsSDK::Core)
set(SUITES buffer texture render draw vertex binding depth timestamp)
set(SOURCES VulkanTests VulkanTextureTests VulkanRenderPassTests VulkanDrawTests VulkanVertexTests VulkanBindingTests VulkanDepthTests VulkanTimestampTests)
foreach(INDEX RANGE 0 7)
  list(GET SUITES ${INDEX} SUITE)
  list(GET SOURCES ${INDEX} SOURCE)
  add_executable(${SUITE}_consumer ${SOURCE}.cpp)
  target_link_libraries(${SUITE}_consumer PRIVATE GraphicsSDK::Core)
endforeach()
