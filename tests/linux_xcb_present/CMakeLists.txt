cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKXcbPresent LANGUAGES CXX)
find_package(GraphicsSDK CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(gfx_present_proxy SHARED "${root}/tests/linux_xcb_surface/Loader.cpp")
set_target_properties(gfx_present_proxy PROPERTIES OUTPUT_NAME vulkan SOVERSION 1 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/proxy")
target_include_directories(gfx_present_proxy SYSTEM PRIVATE "${root}/.deps/Vulkan-Headers/include" ${XCB_INCLUDE_DIRS})
target_link_libraries(gfx_present_proxy PRIVATE ${CMAKE_DL_LIBS})
add_executable(gfx_xcb_present Probe.cpp)
target_link_libraries(gfx_xcb_present PRIVATE GraphicsSDK::Core PkgConfig::XCB ${CMAKE_DL_LIBS})
foreach(target gfx_present_proxy gfx_xcb_present)
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
