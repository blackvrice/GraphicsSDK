cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKCreationFault LANGUAGES CXX)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(GFX_IMPLEMENTATION_ROOT "${root}" CACHE PATH "Product source snapshot")
set(GFX_INSTALLED_SDK "${root}/out/vulkan-creation-install" CACHE PATH "Matching installed SDK")
find_package(GraphicsSDK CONFIG REQUIRED PATHS "${GFX_INSTALLED_SDK}" NO_DEFAULT_PATH)
add_executable(vulkan_creation_fault main.cpp)
target_include_directories(vulkan_creation_fault PRIVATE "${root}/.deps/Vulkan-Headers/include")
target_link_libraries(vulkan_creation_fault PRIVATE GraphicsSDK::Core user32)
target_compile_features(vulkan_creation_fault PRIVATE cxx_std_23)
target_compile_options(vulkan_creation_fault PRIVATE /W4 /wd4100 /permissive- /EHsc /utf-8)
target_compile_definitions(vulkan_creation_fault PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN GFX_VULKAN_IMPLEMENTATION="${GFX_IMPLEMENTATION_ROOT}/src/backends/vulkan/Vulkan.cpp")
enable_testing()
add_test(NAME product_creation_faults COMMAND vulkan_creation_fault)
