file(GLOB_RECURSE GFX_FINGERPRINT_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.hpp")
list(APPEND GFX_FINGERPRINT_FILES "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(SORT GFX_FINGERPRINT_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${GFX_FINGERPRINT_FILES})
set(GFX_FINGERPRINT_CONTENT "")
foreach(GFX_SOURCE IN LISTS GFX_FINGERPRINT_FILES)
  file(SHA256 "${GFX_SOURCE}" GFX_FILE_HASH)
  file(RELATIVE_PATH GFX_RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${GFX_SOURCE}")
  string(APPEND GFX_FINGERPRINT_CONTENT "${GFX_RELATIVE}|${GFX_FILE_HASH}\n")
endforeach()
string(SHA256 GFX_FINGERPRINT "${GFX_FINGERPRINT_CONTENT}")
add_library(GraphicsSDKCore STATIC src/foundation/Backend.cpp src/foundation/VertexInput.cpp src/foundation/Bindings.cpp src/foundation/RenderPass.cpp src/shader/Shader.cpp)
target_compile_definitions(GraphicsSDKCore PRIVATE GFX_SOURCE_SHA256="${GFX_FINGERPRINT}")
add_library(GraphicsSDK::Core ALIAS GraphicsSDKCore)
set_target_properties(GraphicsSDKCore PROPERTIES EXPORT_NAME Core CXX_EXTENSIONS OFF)
target_compile_features(GraphicsSDKCore PUBLIC cxx_std_23)
target_include_directories(GraphicsSDKCore PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(GFX_ENABLE_D3D12)
  if(NOT WIN32)
    message(FATAL_ERROR "D3D12 requires Windows")
  endif()
  target_sources(GraphicsSDKCore PRIVATE src/backends/d3d12/D3D12.cpp)
  target_compile_definitions(GraphicsSDKCore PRIVATE GFX_HAS_D3D12=1)
  target_link_libraries(GraphicsSDKCore PRIVATE d3d12 dxgi dxguid d3dcompiler)
endif()
if(GFX_ENABLE_VULKAN)
  if(NOT EXISTS "${GFX_VULKAN_HEADERS}/include/vulkan/vulkan_core.h")
    message(FATAL_ERROR "Vulkan requires Vulkan-Headers; set GFX_VULKAN_HEADERS (see docs/VULKAN.md)")
  endif()
  target_sources(GraphicsSDKCore PRIVATE src/backends/vulkan/Vulkan.cpp)
  target_include_directories(GraphicsSDKCore SYSTEM PRIVATE "${GFX_VULKAN_HEADERS}/include")
  target_compile_definitions(GraphicsSDKCore PRIVATE GFX_HAS_VULKAN=1)
  if(GFX_VULKAN_XCB)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GFX_XCB REQUIRED xcb)
    target_include_directories(GraphicsSDKCore SYSTEM PRIVATE ${GFX_XCB_INCLUDE_DIRS})
    target_compile_definitions(GraphicsSDKCore PRIVATE GFX_HAS_VULKAN_XCB=1)
  endif()
  target_link_libraries(GraphicsSDKCore PRIVATE ${CMAKE_DL_LIBS})
  if(WIN32)
    target_link_libraries(GraphicsSDKCore PRIVATE user32)
  endif()
endif()
if(MSVC)
  target_compile_options(GraphicsSDKCore PRIVATE /W4 /permissive- /utf-8)
  target_compile_definitions(GraphicsSDKCore PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(GraphicsSDKCore PRIVATE -Wall -Wextra -Wpedantic)
endif()
