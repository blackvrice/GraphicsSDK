cmake_minimum_required(VERSION 3.25)
project(GraphicsSDK VERSION 0.0.1 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(GFX_BUILD_TESTS "Build CPU and GPU correctness tests" ON)
option(GFX_BUILD_EXAMPLES "Build standalone examples" ON)
option(GFX_ENABLE_D3D12 "Build the D3D12 backend" ${WIN32})
file(GLOB_RECURSE GFX_FINGERPRINT_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
list(APPEND GFX_FINGERPRINT_FILES "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(SORT GFX_FINGERPRINT_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${GFX_FINGERPRINT_FILES})
set(GFX_FINGERPRINT_CONTENT "")
foreach(GFX_SOURCE IN LISTS GFX_FINGERPRINT_FILES)
  file(SHA256 "${GFX_SOURCE}" GFX_FILE_HASH)
  file(RELATIVE_PATH GFX_RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${GFX_SOURCE}")
  string(APPEND GFX_FINGERPRINT_CONTENT "${GFX_RELATIVE}|${GFX_FILE_HASH}\n")
endforeach()
string(SHA256 GFX_FINGERPRINT "${GFX_FINGERPRINT_CONTENT}")
add_library(GraphicsSDKCore STATIC src/foundation/Backend.cpp src/foundation/VertexInput.cpp src/foundation/Bindings.cpp src/foundation/RenderPass.cpp src/shader/Shader.cpp)
target_compile_definitions(GraphicsSDKCore PRIVATE GFX_SOURCE_SHA256="${GFX_FINGERPRINT}")
add_library(GraphicsSDK::Core ALIAS GraphicsSDKCore)
set_target_properties(GraphicsSDKCore PROPERTIES EXPORT_NAME Core CXX_EXTENSIONS OFF)
target_compile_features(GraphicsSDKCore PUBLIC cxx_std_23)
target_include_directories(GraphicsSDKCore PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(GFX_ENABLE_D3D12)
  if(NOT WIN32)
    message(FATAL_ERROR "D3D12 requires Windows")
  endif()
  target_sources(GraphicsSDKCore PRIVATE src/backends/d3d12/D3D12.cpp)
  target_compile_definitions(GraphicsSDKCore PRIVATE GFX_HAS_D3D12=1)
  target_link_libraries(GraphicsSDKCore PRIVATE d3d12 dxgi dxguid d3dcompiler)
endif()
if(MSVC)
  target_compile_options(GraphicsSDKCore PRIVATE /W4 /permissive- /utf-8)
  target_compile_definitions(GraphicsSDKCore PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(GraphicsSDKCore PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(GFX_BUILD_TESTS)
  enable_testing()
  if(WIN32)
    add_executable(gfx_debug_capture tools/DebugCapture.cpp)
    target_compile_features(gfx_debug_capture PRIVATE cxx_std_23)
  endif()
  add_executable(gfx_tests tests/CoreTests.cpp)
  target_link_libraries(gfx_tests PRIVATE GraphicsSDK::Core)
  add_test(NAME core_contracts COMMAND gfx_tests --cpu)
  add_executable(gfx_vertex_tests tests/VertexInputTests.cpp)
  target_link_libraries(gfx_vertex_tests PRIVATE GraphicsSDK::Core)
  add_test(NAME vertex_layout_contracts COMMAND gfx_vertex_tests --cpu)
  add_executable(gfx_binding_tests tests/BindingTests.cpp)
  target_link_libraries(gfx_binding_tests PRIVATE GraphicsSDK::Core)
  add_test(NAME binding_layout_contracts COMMAND gfx_binding_tests --cpu)
  add_executable(gfx_render_tests tests/RenderPassTests.cpp)
  target_link_libraries(gfx_render_tests PRIVATE GraphicsSDK::Core)
  add_test(NAME render_pass_contracts COMMAND gfx_render_tests --cpu)
  if(GFX_ENABLE_D3D12)
    add_executable(gfx_recovery_tests tests/RecoveryTests.cpp)
    target_link_libraries(gfx_recovery_tests PRIVATE GraphicsSDK::Core)
    add_test(NAME d3d12_hardware COMMAND gfx_tests --gpu)
    add_test(NAME d3d12_validation COMMAND gfx_tests --validation)
    add_test(NAME d3d12_warp COMMAND gfx_tests --warp)
    set_tests_properties(d3d12_hardware d3d12_validation d3d12_warp PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)
    foreach(mode gpu validation warp)
      add_test(NAME vertex_input_${mode} COMMAND gfx_vertex_tests --${mode})
      set_tests_properties(vertex_input_${mode} PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
      add_test(NAME resource_bindings_${mode} COMMAND gfx_binding_tests --${mode})
      set_tests_properties(resource_bindings_${mode} PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
      add_test(NAME render_pass_${mode} COMMAND gfx_render_tests --${mode})
      set_tests_properties(render_pass_${mode} PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
      add_test(NAME device_recovery_${mode} COMMAND gfx_recovery_tests --${mode})
      set_tests_properties(device_recovery_${mode} PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
    endforeach()
  endif()
endif()
if(GFX_BUILD_EXAMPLES)
  add_executable(gfx_triangle examples/Triangle.cpp)
  target_link_libraries(gfx_triangle PRIVATE GraphicsSDK::Core)
  add_executable(gfx_indexed_quads examples/IndexedQuads.cpp)
  target_link_libraries(gfx_indexed_quads PRIVATE GraphicsSDK::Core)
endif()
install(TARGETS GraphicsSDKCore EXPORT GraphicsSDKTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/graphics DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT GraphicsSDKTargets NAMESPACE GraphicsSDK:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GraphicsSDK)
configure_package_config_file(cmake/GraphicsSDKConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/GraphicsSDKConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GraphicsSDK)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/GraphicsSDKConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/GraphicsSDKConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/GraphicsSDKConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GraphicsSDK)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/GraphicsSDK)
