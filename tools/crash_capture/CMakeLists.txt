cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKCrashCapture LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Crash capture requires Windows x64")
endif()
add_executable(gfx_crash_capture Capture.cpp)
target_link_libraries(gfx_crash_capture PRIVATE dbghelp)
add_executable(gfx_crash_fixture Fixture.cpp)
get_filename_component(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(GFX_CRASH_CORE_LIBRARY "${SDK_ROOT}/out/core/Release/GraphicsSDKCore.lib" CACHE FILEPATH "Exact SDK library used by the symbol workload")
if(NOT EXISTS "${GFX_CRASH_CORE_LIBRARY}")
  message(FATAL_ERROR "Build the selected SDK library first: ${GFX_CRASH_CORE_LIBRARY}")
endif()
add_executable(gfx_memory_stress_symbols "${SDK_ROOT}/tools/MemoryStress.cpp")
target_include_directories(gfx_memory_stress_symbols PRIVATE "${SDK_ROOT}/include")
target_link_libraries(gfx_memory_stress_symbols PRIVATE "${GFX_CRASH_CORE_LIBRARY}" d3d12 dxgi dxguid d3dcompiler psapi)
foreach(target gfx_crash_capture gfx_crash_fixture gfx_memory_stress_symbols)
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /Zi)
  target_link_options(${target} PRIVATE /DEBUG)
endforeach()
