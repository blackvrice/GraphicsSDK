cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKFixedMemoryWork LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Fixed memory diagnostic requires Windows x64")
endif()
get_filename_component(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(gfx_memory_fixed_work FixedWork.cpp)
target_compile_features(gfx_memory_fixed_work PRIVATE cxx_std_23)
target_compile_options(gfx_memory_fixed_work PRIVATE /W4 /permissive- /utf-8 /Zi)
target_link_options(gfx_memory_fixed_work PRIVATE /DEBUG)
target_include_directories(gfx_memory_fixed_work PRIVATE "${SDK_ROOT}/include")
target_link_libraries(gfx_memory_fixed_work PRIVATE "${SDK_ROOT}/out/core/Release/GraphicsSDKCore.lib" d3d12 dxgi dxguid d3dcompiler psapi)
