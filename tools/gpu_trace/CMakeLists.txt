cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKGpuTrace LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows ETW tools only")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(gpu_etl_export Export.cpp)
target_link_libraries(gpu_etl_export PRIVATE tdh advapi32)
add_executable(gpu_memory_control Control.cpp "${REPO}/benchmarks/common/Environment.cpp" "${REPO}/benchmarks/common/Telemetry.cpp")
target_include_directories(gpu_memory_control PRIVATE "${REPO}/benchmarks" "${REPO}/.deps/LLGL/include")
target_link_libraries(gpu_memory_control PRIVATE "${REPO}/out/llgl/build/Release/LLGL.lib" d3d12 dxgi advapi32)
foreach(target gpu_etl_export gpu_memory_control)
  target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
endforeach()
foreach(dll LLGL LLGL_Direct3D12)
  add_custom_command(TARGET gpu_memory_control POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${REPO}/out/llgl/build/Release/${dll}.dll" "$<TARGET_FILE_DIR:gpu_memory_control>")
endforeach()
enable_testing()
add_test(NAME gpu_memory_positive_control COMMAND gpu_memory_control)
set_tests_properties(gpu_memory_positive_control PROPERTIES RUN_SERIAL TRUE TIMEOUT 120)
