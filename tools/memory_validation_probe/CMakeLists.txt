cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKMemoryValidationProbe LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This diagnostic requires Windows D3D12 Graphics Tools")
endif()
get_filename_component(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(memory_validation_probe main.cpp)
target_compile_features(memory_validation_probe PRIVATE cxx_std_23)
target_include_directories(memory_validation_probe PRIVATE "${SDK_ROOT}/include")
target_link_libraries(memory_validation_probe PRIVATE "${SDK_ROOT}/out/core/Release/GraphicsSDKCore.lib" d3d12 dxgi dxguid d3dcompiler psapi)
target_compile_options(memory_validation_probe PRIVATE /W4 /permissive- /utf-8)
add_executable(gbv_lifetime_probe Lifetime.cpp)
target_compile_features(gbv_lifetime_probe PRIVATE cxx_std_23)
target_link_libraries(gbv_lifetime_probe PRIVATE d3d12 dxgi dxguid psapi)
target_compile_options(gbv_lifetime_probe PRIVATE /W4 /permissive- /utf-8)
add_executable(sdk_lifetime_probe SDKLifetime.cpp)
target_compile_features(sdk_lifetime_probe PRIVATE cxx_std_23)
target_include_directories(sdk_lifetime_probe PRIVATE "${SDK_ROOT}/include")
target_link_libraries(sdk_lifetime_probe PRIVATE "${SDK_ROOT}/out/core/Release/GraphicsSDKCore.lib" d3d12 dxgi dxguid d3dcompiler psapi)
target_compile_options(sdk_lifetime_probe PRIVATE /W4 /permissive- /utf-8)
add_executable(gbv_runtime_probe Runtime.cpp)
target_compile_features(gbv_runtime_probe PRIVATE cxx_std_23)
target_link_libraries(gbv_runtime_probe PRIVATE d3d12 dxgi dxguid psapi)
target_compile_options(gbv_runtime_probe PRIVATE /W4 /permissive- /utf-8)
