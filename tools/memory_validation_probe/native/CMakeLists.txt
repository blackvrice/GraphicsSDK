cmake_minimum_required(VERSION 3.25)
project(GBVLifetimeReproduction LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This reproduction requires Windows D3D12 Graphics Tools")
endif()
add_executable(gbv_lifetime_probe ../Lifetime.cpp)
target_compile_features(gbv_lifetime_probe PRIVATE cxx_std_23)
target_link_libraries(gbv_lifetime_probe PRIVATE d3d12 dxgi dxguid psapi)
target_compile_options(gbv_lifetime_probe PRIVATE /W4 /permissive- /utf-8)
add_executable(gbv_runtime_probe ../Runtime.cpp)
target_compile_features(gbv_runtime_probe PRIVATE cxx_std_23)
target_link_libraries(gbv_runtime_probe PRIVATE d3d12 dxgi dxguid psapi)
target_compile_options(gbv_runtime_probe PRIVATE /W4 /permissive- /utf-8)
