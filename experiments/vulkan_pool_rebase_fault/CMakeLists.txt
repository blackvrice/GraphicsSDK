cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKRebasedPoolFault LANGUAGES CXX)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(GFX_INSTALLED_SDK "${root}/out/vulkan-pool-churn-install" CACHE PATH "Exact reviewed rebased candidate")
find_package(GraphicsSDK CONFIG REQUIRED PATHS "${GFX_INSTALLED_SDK}" NO_DEFAULT_PATH)
set(own "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(APPEND own "${root}/experiments/vulkan_pool_churn/Layer.cpp" "${root}/experiments/vulkan_pool_churn/Snapshot.hpp" "${root}/experiments/vulkan_buffer_pool_fault/Layer.cpp" "${root}/experiments/vulkan_buffer_pool_fault/Accounting.hpp" "${root}/experiments/vulkan_buffer_pool_fault/FaultPlan.hpp" "${root}/experiments/vulkan_buffer_pool_fault/Client.hpp" "${root}/benchmarks/common/Adapter.hpp" "${root}/benchmarks/common/Environment.cpp" "${root}/benchmarks/vulkan_upload/Identity.hpp")
list(SORT own)
set(identity "")
foreach(p IN LISTS own)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${p}")
  file(SHA256 "${p}" sha)
  file(RELATIVE_PATH relative "${root}" "${p}")
  string(APPEND identity "${relative}|${sha}\n")
endforeach()
string(SHA256 fingerprint "${identity}")
add_executable(vulkan_rebase_fault main.cpp "${root}/benchmarks/common/Environment.cpp")
target_link_libraries(vulkan_rebase_fault PRIVATE GraphicsSDK::Core dxgi advapi32)
add_library(VkLayer_gfx_buffer_fault SHARED "${root}/experiments/vulkan_pool_churn/Layer.cpp")
foreach(target vulkan_rebase_fault VkLayer_gfx_buffer_fault)
  target_include_directories(${target} PRIVATE "${root}/.deps/Vulkan-Headers/include")
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN GFX_TRACE_FINGERPRINT="${fingerprint}")
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/VkLayer_gfx_buffer_fault.json" CONTENT "{\"file_format_version\":\"1.2.0\",\"layer\":{\"name\":\"VK_LAYER_GFX_buffer_fault\",\"type\":\"GLOBAL\",\"library_path\":\"./VkLayer_gfx_buffer_fault.dll\",\"api_version\":\"1.1.0\",\"implementation_version\":\"1\",\"functions\":{\"vkNegotiateLoaderLayerInterfaceVersion\":\"gfxNegotiateLayer\"},\"description\":\"GraphicsSDK mixed-context exact-size fallback contracts\"}}\n")
