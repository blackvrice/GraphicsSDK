cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKUploadRingRanges LANGUAGES CXX)

enable_testing()
add_executable(upload_ring_ranges RangeTests.cpp)
target_include_directories(upload_ring_ranges PRIVATE ../../include)
target_compile_features(upload_ring_ranges PRIVATE cxx_std_23)
set_target_properties(upload_ring_ranges PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(upload_ring_ranges PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(upload_ring_ranges PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME upload_ring_ranges COMMAND upload_ring_ranges)
