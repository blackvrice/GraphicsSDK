cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKVulkanBufferFault LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This fault-injection profile requires Windows x64 MSVC")
endif()
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(GFX_INSTALLED_SDK "${root}/out/vulkan-buffer-pool-fault-install" CACHE PATH "Isolated SDK installation")
find_package(GraphicsSDK CONFIG REQUIRED PATHS "${GFX_INSTALLED_SDK}" NO_DEFAULT_PATH)
file(GLOB own CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
list(APPEND own "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${root}/benchmarks/common/Adapter.hpp" "${root}/benchmarks/common/Environment.cpp" "${root}/benchmarks/vulkan_upload/Identity.hpp")
list(SORT own)
set(identity "")
foreach(p IN LISTS own)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${p}")
  file(SHA256 "${p}" sha)
  file(RELATIVE_PATH relative "${root}" "${p}")
  string(APPEND identity "${relative}|${sha}\n")
endforeach()
string(SHA256 fingerprint "${identity}")
add_executable(vulkan_buffer_fault main.cpp "${root}/benchmarks/common/Environment.cpp")
target_link_libraries(vulkan_buffer_fault PRIVATE GraphicsSDK::Core dxgi advapi32)
add_library(VkLayer_gfx_buffer_fault SHARED Layer.cpp)
add_executable(vulkan_fault_plan PlanTests.cpp)
foreach(target vulkan_buffer_fault VkLayer_gfx_buffer_fault vulkan_fault_plan)
  target_include_directories(${target} PRIVATE "${root}/.deps/Vulkan-Headers/include")
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN GFX_TRACE_FINGERPRINT="${fingerprint}")
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
endforeach()
enable_testing()
add_test(NAME fault_plan_contracts COMMAND vulkan_fault_plan)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/VkLayer_gfx_buffer_fault.json" CONTENT "{\"file_format_version\":\"1.2.0\",\"layer\":{\"name\":\"VK_LAYER_GFX_buffer_fault\",\"type\":\"GLOBAL\",\"library_path\":\"./VkLayer_gfx_buffer_fault.dll\",\"api_version\":\"1.1.0\",\"implementation_version\":\"1\",\"functions\":{\"vkNegotiateLoaderLayerInterfaceVersion\":\"gfxNegotiateLayer\"},\"description\":\"GraphicsSDK guarded synthetic buffer failure tests\"}}\n")
