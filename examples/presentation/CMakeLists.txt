cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKPresentation LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This Win32 example requires Windows and MSVC")
endif()
# Deliberately consume an installed SDK, just as a separate application would.
find_package(GraphicsSDK CONFIG REQUIRED)
add_executable(gfx_presentation Windowed.cpp app.manifest)
target_link_libraries(gfx_presentation PRIVATE GraphicsSDK::Core user32)
target_compile_definitions(gfx_presentation PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(gfx_presentation PRIVATE /W4 /permissive- /utf-8)
include(CTest)
if(BUILD_TESTING)
  add_executable(gfx_presentation_policy_tests PolicyTests.cpp)
  target_compile_features(gfx_presentation_policy_tests PRIVATE cxx_std_23)
  target_compile_options(gfx_presentation_policy_tests PRIVATE /W4 /permissive- /utf-8)
  add_test(NAME presentation_policies COMMAND gfx_presentation_policy_tests)
  foreach(mode gpu validation warp)
    foreach(sync vsync immediate)
      set(extra_args)
      if(sync STREQUAL "immediate")
        set(extra_args --no-vsync)
      endif()
      add_test(NAME presentation_${mode}_${sync} COMMAND gfx_presentation --self-test --mode ${mode} ${extra_args})
      set_tests_properties(presentation_${mode}_${sync} PROPERTIES RUN_SERIAL TRUE TIMEOUT 90)
    endforeach()
  endforeach()
endif()
