cmake_minimum_required(VERSION 3.25)
project(GraphicsSDKXcbWindow LANGUAGES CXX)
if(NOT UNIX OR APPLE)
  message(FATAL_ERROR "The XCB example requires Linux")
endif()
find_package(GraphicsSDK CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
add_executable(gfx_xcb_window Main.cpp)
target_link_libraries(gfx_xcb_window PRIVATE GraphicsSDK::Core PkgConfig::XCB)
target_compile_features(gfx_xcb_window PRIVATE cxx_std_23)
target_compile_options(gfx_xcb_window PRIVATE -Wall -Wextra -Wpedantic)
add_executable(gfx_xcb_policy_tests PolicyTests.cpp)
target_compile_features(gfx_xcb_policy_tests PRIVATE cxx_std_23)
enable_testing()
add_test(NAME xcb_lifecycle_policy COMMAND gfx_xcb_policy_tests)
